import discord
import random

intents = discord.Intents.default()
intents.message_content = True

client = discord.Client(intents=intents)

@client.event
async def on_ready():
    print(f'We have logged in as {client.user}')

chosen_joke = None
def jokes(joke_number):
    if joke_number == 1:
        chosen_joke == "joke 1"
    if joke_number == 2:
        chosen_joke == "joke 2"
    if joke_number == 3:
        chosen_joke == "joke 3"
    if joke_number == 4:
        chosen_joke == "joke 4"
    if joke_number == 5:
        chosen_joke == "joke 5"
    else:
        chosen_joke == "Hur?"
    return chosen_joke

@client.event
async def on_message(message):
    if message.author == client.user:
        return

    if message.content.startswith('!introduktion'):
        await message.channel.send('jag är emo svampbob')
    
    if message.content.startswith('!help'):
        await message.channel.send('!introduktion \n !roll \n !skämt')

    if message.content.startswith('!roll'):
        parts = message.content.split()

        if len(parts) > 1: #nödvändig?
            arg = parts[1] #förstår inte helt

            try:
                max_number = int(arg)
                random_number = random.randint(1, max_number)
                await message.channel.send(f"You rolled {random_number}")
            except ValueError:
                await message.channel.send(f"Enter a valid number")

        else:
            random_number = random.randint(1, 20)
            await message.channel.send(f"You rolled {random_number}!")

    if message.content.startswith('!skämt'):
        joke_number = random.randint(1,5)
        chosen_joke = jokes(joke_number)
        await message.channel.send(chosen_joke)
